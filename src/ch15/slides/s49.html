Printing the reference count

Listing 15.19: Printing the reference count — src/main.rs

// --snip--

 

fn main() {

    let a = Rc::new(Cons(5, Rc::new(Cons(10, Rc::new(Nil)))));

    println!("count after creating a = {}", Rc::strong_count(&a));

    let b = Cons(3, Rc::clone(&a));

    println!("count after creating b = {}", Rc::strong_count(&a));

    {

        let c = Cons(4, Rc::clone(&a));

        println!("count after creating c = {}", Rc::strong_count(&a));

    }

    println!("count after c goes out of scope = {}", Rc::strong_count(&a));

}

At each point where the count changes, we print it with Rc::strong_count. It is named strong_count rather than count because Rc<T> also has a weak_count, explained in “Preventing Reference Cycles Using Weak<T>” (15.6).
