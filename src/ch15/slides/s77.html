Chapter 15 · 15.5 RefCell<T> and the Interior Mutability Pattern

Mutating a value shared by three lists

Listing 15.24 (continued): main — src/main.rs

fn main() {

    let value = Rc::new(RefCell::new(5));

 

    let a = Rc::new(Cons(Rc::clone(&value), Rc::new(Nil)));

 

    let b = Cons(Rc::new(RefCell::new(3)), Rc::clone(&a));

    let c = Cons(Rc::new(RefCell::new(4)), Rc::clone(&a));

 

    *value.borrow_mut() += 10;

 

    println!("a after = {a:?}");

    println!("b after = {b:?}");

    println!("c after = {c:?}");

}
