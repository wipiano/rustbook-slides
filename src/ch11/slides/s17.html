Asserting a negative: smaller cannot hold larger

It does pass! Now add a test asserting that a smaller rectangle cannot hold a larger rectangle:

src/lib.rs

#[cfg(test)]

mod tests {

    use super::*;

 

    #[test]

    fn larger_can_hold_smaller() {

        // --snip--

    }

 

    #[test]

    fn smaller_cannot_hold_larger() {

        let larger = Rectangle {

            width: 8,

            height: 7,

        };

        let smaller = Rectangle {

            width: 5,

            height: 1,

        };

 

        assert!(!smaller.can_hold(&larger));

    }

}

The correct result of can_hold here is false, so we negate it before passing it to assert!. As a result, the test passes if can_hold returns false.
