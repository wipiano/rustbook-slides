Adding a test to a workspace

For another enhancement, add a test of the add_one::add_one function within the add_one crate:

add_one/src/lib.rs

pub fn add_one(x: i32) -> i32 {

    x + 1

}

 

#[cfg(test)]

mod tests {

    use super::*;

 

    #[test]

    fn it_works() {

        assert_eq!(3, add_one(2));

    }

}
