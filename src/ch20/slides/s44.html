Chapter 20 · 20.2 Advanced Traits

Two associated functions named baby_name

Listing 20.20: A trait with an associated function and a type with an associated function of the same name that also implements the trait — src/main.rs

trait Animal {

    fn baby_name() -> String;

}

 

struct Dog;

 

impl Dog {

    fn baby_name() -> String {

        String::from("Spot")

    }

}

 

impl Animal for Dog {

    fn baby_name() -> String {

        String::from("puppy")

    }

}

fn main() {

    println!("A baby dog is called a {}", Dog::baby_name());

}
