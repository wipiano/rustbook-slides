Satisfying the supertrait: implement Display

To fix this, implement Display on Point (src/main.rs; trait and struct unchanged):

impl OutlinePrint for Point {}

 

use std::fmt;

 

impl fmt::Display for Point {

    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {

        write!(f, "({}, {})", self.x, self.y)

    }

}

 

fn main() {

    let p = Point { x: 1, y: 3 };

    p.outline_print();

}

Now implementing OutlinePrint on Point compiles, and outline_print displays a Point within an outline of asterisks.
