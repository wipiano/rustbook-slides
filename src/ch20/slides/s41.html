Chapter 20 · 20.2 Advanced Traits

Three methods named fly

Listing 20.17: Two traits are defined to have a fly method and are implemented on the Human type, and a fly method is implemented on Human directly. — src/main.rs

trait Pilot {

    fn fly(&self);

}

 

trait Wizard {

    fn fly(&self);

}

 

struct Human;

 

impl Pilot for Human {

    fn fly(&self) {

        println!("This is your captain speaking.");

    }

}

 

impl Wizard for Human {

    fn fly(&self) {

        println!("Up!");

    }

}

 

impl Human {

    fn fly(&self) {

        println!("*waving arms furiously*");

    }

}

 

fn main() {}
