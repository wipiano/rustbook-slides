Awaiting an async block inside block_on

Listing 17.4: Awaiting an async block with trpl::block_on — src/main.rs

use trpl::Html;

 

fn main() {

    let args: Vec<String> = std::env::args().collect();

 

    trpl::block_on(async {

        let url = &args[1];

        match page_title(url).await {

            Some(title) => println!("The title for {url} was {title}"),

            None => println!("{url} had no title"),

        }

    })

}

main stays an ordinary function; the async block passed to trpl::block_on is where await is allowed. page_title follows on the next slide.
