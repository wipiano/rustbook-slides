Chapter 18 · 18.3 Implementing an Object-Oriented Design Pattern

Post and DraftPost as separate types

Listing 18.19: A Post with a content method and a DraftPost without a content method — src/lib.rs

pub struct Post {

    content: String,

}

 

pub struct DraftPost {

    content: String,

}

 

impl Post {

    pub fn new() -> DraftPost {

        DraftPost {

            content: String::new(),

        }

    }

 

    pub fn content(&self) -> &str {

        &self.content

    }

}

 

impl DraftPost {

    pub fn add_text(&mut self, text: &str) {

        self.content.push_str(text);

    }

}
