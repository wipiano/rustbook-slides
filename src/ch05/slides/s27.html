Opting in with #[derive(Debug)]

Listing 5.12: Adding the attribute to derive the Debug trait and printing the Rectangle instance using debug formatting — src/main.rs

#[derive(Debug)]

struct Rectangle {

    width: u32,

    height: u32,

}

 

fn main() {

    let rect1 = Rectangle {

        width: 30,

        height: 50,

    };

 

    println!("rect1 is {rect1:?}");

}
