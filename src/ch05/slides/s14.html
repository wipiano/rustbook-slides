Storing references without lifetimes fails

This src/main.rs won’t compile: username and email are references (&str) with no lifetimes specified.

struct User {

    active: bool,

    username: &str,

    email: &str,

    sign_in_count: u64,

}

 

fn main() {

    let user1 = User {

        active: true,

        username: "someusername123",

        email: "someone@example.com",

        sign_in_count: 1,

    };

}
