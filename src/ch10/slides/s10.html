Listing 10.4: Two functions that differ only in their names and in the types in their signatures — src/main.rs

fn largest_i32(list: &[i32]) -> &i32 {

    let mut largest = &list[0];

 

    for item in list {

        if item > largest {

            largest = item;

        }

    }

 

    largest

}

fn largest_char(list: &[char]) -> &char {

            largest = item;

        }

    }

 

    largest

}
